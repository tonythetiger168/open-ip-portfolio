LINE case:DDR_top \| default: dstate <= D_IDLE; # unused 3-bit encoding of the 7-state enum
LINE cond_else:DDR_top \| wire \[CHW-1:0\] c_ch = \(NCH == 1\) # NCH=1 elaboration constant: one channel, tied to 0
LINE else:DDR_top \| if \(act_ok\) begin # unreachable: D_ACT is entered only after the PRE path waited tRAS then tRP (17 clocks) >= tRC (17) since the bank's last ACT, so act_ok always holds there (tRC itself is checked by a_trc)
TOGGLE DDR_top\.addr\[1[4-6]\]: # always 0: rows are 14 bits, columns 8 bits
TOGGLE DDR_top\.trace_cmd\[2\]: # constant 0: trace codes are 1 ACT, 2 RD, 3 WR
TOGGLE DDR_top\.dqs\[[01]\]: # tied Hi-Z: the model has no strobe
TOGGLE DDR_top\.rp_t\[\d+\]\[(3|4)\]: # constant 0: largest load is tRP = 5 < 8
TOGGLE DDR_top\.ras_t\[\d+\]\[(4)\]: # constant 0: largest load is tRAS = 12 < 16
TOGGLE DDR_top\.c_ch\[0\]: # NCH=1: channel field tied to 0
