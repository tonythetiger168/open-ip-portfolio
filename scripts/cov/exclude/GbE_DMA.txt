# Store and forward: a TX frame is handed to the MAC only once it is whole in
# the TX FIFO, and the MAC takes a byte per byte time while the FIFO always
# has the next one, so the MAC never underruns (a_tx_store_fwd checks the
# FIFO holds the rest of the frame whenever TX_VALID is high).
TOGGLE _top\.(tx_underrun|tx_ur): # the MAC never underruns: store and forward
LINE GbE_DMA_top\.sv:226:if: # the MAC never underruns: store and forward
TOGGLE _top\.gtx_er: # GTX_ER is only sent on a TX underrun (store and forward: never)
# The scheduler arms for S_TXD and S_RXD are one case item; a TX descriptor
# always goes on to S_TXB or back to S_IDLE (st is checked by a_sched and the
# job assertions), so the RX next states are never taken from S_TXD.
FSM fsm_arc__st__S_TXD__S_RX(B|DRAIN)$ # shared S_TXD / S_RXD case item: RX next states only from S_RXD
LINE GbE_DMA_top\.sv:291:case: # st holds the 8 encoded steps only
TOGGLE _top\.st\[3\]: # st holds the 8 encoded steps only (0..7)
# The MAC passes on no frame shorter than 2 bytes (a line frame that ends
# before its sixth byte is a fragment), so a frame is never last on its first
# byte and a dropped frame has always started.
LINE GbE_DMA_top\.sv:334:cond_else: # no 1-byte frames from the MAC
# rx_pk holds bytes 0..2 of a word; the fourth byte goes straight into the FIFO
# word, so rx_pk[31:24] is only ever cleared.
TOGGLE _top\.rx_pk\[(2[4-9]|3[01])\]: # byte 3 is never held in the packer
